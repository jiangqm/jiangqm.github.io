粒子系统

Canvas

Canvas 2D 粒子系统动画，鼠标移动产生轨迹粒子，点击产生爆炸效果。支持重力、衰减等参数调节。

参数控制

粒子数量100
速度2
大小3
颜色
重力
衰减0.98

粒子系统核心代码

"use client"

// 粒子对象池 — 预分配固定数组，避免 GC
const pool: Particle[] = Array.from({ length: 500 }, () => ({
  x: 0, y: 0, vx: 0, vy: 0,
  life: 0, maxLife: 0, size: 0,
  color: "", active: false,
}))

// 从池中获取一个非活跃粒子
function createParticle(pool: Particle[]) {
  return pool.find(p => !p.active) ?? null
}

// 鼠标移动 → 轨迹粒子
function spawnTrail(pool, x, y, params) {
  const p = createParticle(pool)
  if (!p) return
  const angle = Math.random() * Math.PI * 2
  p.x = x; p.y = y
  p.vx = Math.cos(angle) * params.speed * 0.5
  p.vy = Math.sin(angle) * params.speed * 0.5
  p.life = 1; p.maxLife = 60 + Math.random() * 40
  p.size = params.size * (0.5 + Math.random() * 0.5)
  p.color = params.color; p.active = true
}

// 鼠标点击 → 爆炸粒子
function spawnExplosion(pool, x, y, params) {
  for (let i = 0; i < params.count; i++) {
    const p = createParticle(pool)
    if (!p) break
    const angle = (Math.PI * 2 * i) / params.count
    p.x = x; p.y = y
    p.vx = Math.cos(angle) * params.speed * 2
    p.vy = Math.sin(angle) * params.speed * 2
    p.life = 1; p.maxLife = 40 + Math.random() * 60
    p.color = params.color; p.active = true
  }
}

// 动画循环
function animate(ctx, width, height, pool, params) {
  ctx.clearRect(0, 0, width, height)
  for (const p of pool) {
    if (!p.active) continue
    p.life -= 1 / p.maxLife
    if (p.life <= 0) { p.active = false; continue }
    if (params.gravity) p.vy += 0.05
    p.vx *= params.decay
    p.vy *= params.decay
    p.x += p.vx; p.y += p.vy
    const alpha = p.life
    ctx.globalAlpha = alpha
    ctx.fillStyle = p.color
    ctx.beginPath()
    ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2)
    ctx.fill()
  }
  ctx.globalAlpha = 1
}
